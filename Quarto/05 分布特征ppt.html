添加核密度曲线

代码 
p4 <- ggplot(Wage)+
  geom_histogram(aes(x=wage,y = after_stat(density)),
                 fill="lightskyblue",color="grey50",
                 binwidth = 10)+
  geom_density(aes(x=wage),color="red")+
  annotate("text",x=200,y=0.005,label="核密度曲线",family="STKaiti")+
  annotate("segment",x=170,xend = 160,y=0.005,yend = 0.005,
           arrow=arrow(angle = 15,length = unit(0.1,"inches")))

添加理论正态分布曲线和均值线

代码 
p5 <- ggplot(Wage)+
  geom_histogram(aes(x=wage,y = after_stat(density)),
                 fill="lightskyblue",color="grey50",
                 binwidth = 10)+
  stat_function(fun = dnorm,args = list(mean=mean(Wage$wage),sd=sd(Wage$wage)),
  linetype=5,color="red2")+
  annotate("text",x=230,y=0.0025,label="理论正态分布曲线",family="STKaiti")+
  geom_vline(xintercept = mean(Wage$wage),linetype=2,
             color="red2")+
  annotate("text",x=125,y=0.012,label="均值线",family="STKaiti")
