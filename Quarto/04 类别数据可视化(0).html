funy <- function(x) {
  s_mean=mean(x)
  s_sd=sd(x)
  s_n=length(x)
  s_sem=sd(x)/sqrt(length(x))
  return(c(s_mean,s_sd,s_n,s_sem))
}

d3 <- data2_1 %>% gather(key="课程",value = "分数",R语言,Python语言) %>% 
  group_by(课程) %>% 
  mutate(mean=mean(分数),

         n=length(分数),
         sem=sd(分数)/sqrt(n)) %>% 
  select(课程,mean,sd,n,sem) %>% unique()

  s_sd=sd(x)
