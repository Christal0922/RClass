Code

ggplot(data4_1,aes(x=AQI))+

2.3 计算分布相关参数

2.3.1 计算偏态系数

Code 
library(e1071)
l <- round(skewness(data4_1$AQI),4)

paste0实现字符串拼接

Code 
a <- paste0("偏态系数=",l)
a
[1] "偏态系数=1.1986"

直方图中添加偏态系数信息

Code 
p1 <- ggplot(data4_1,aes(x=AQI))+
  geom_histogram(fill="lightgreen",
                 color="grey")
p1
Code 
p1+annotate("text",x=210,y=37,
            label=a)
