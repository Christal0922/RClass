2.2.5.2 plyr包的mutate函数计算平均成绩和标准差

Code 
df4 <- data2_1 %>% melt(id.vars = c("性别","专业"),
                 variable.name = "课程",
                 value.name = "分数") %>% 
  group_by(课程) %>% 
  mutate(mean=mean(分数),
         sd=sd(分数),
         n=length(分数),
         sem=sd(分数)/sqrt(n)) %>% 
  select(课程,mean,sd) %>% unique()
Code 
ggplot(df4,aes(x=课程,y=mean,fill=课程))+
  geom_col(width = 0.5)+
  geom_errorbar(aes(ymin = mean-sd,
                    ymax = mean+sd),
                width = 0.3)+
  mytheme

2.2.5.3 mutate函数绘制并列条形图

Code 
data2_1 %>% melt(id.vars = c("性别","专业"),
                 variable.name = "课程",
                 value.name = "分数") %>% 
  group_by(课程,性别) %>% 
  mutate(mean=mean(分数),
         sd=sd(分数),
         n=length(分数),
         sem=sd(分数)/sqrt(n)) %>% 
  select(课程,性别,mean,sd) %>% unique() %>% 
  ggplot(aes(x=课程,y=mean,fill=性别))+
  geom_bar(stat = "identity",
           position = "dodge")+
  geom_errorbar(aes(ymin = mean-sd,
                    ymax = mean+sd),
                width = 0.3,
                position = position_dodge(0.9))+
  labs(y="平均成绩")+
  guides(fill=guide_legend(title = NULL))+
  mytheme

  • 设置自定义函数mystats存储统计信息
Code 
mystats <- function(x) {
  mean=mean(x)
  sd=sd(x)
  n=length(x)
  sem=sd(x)/sqrt(n)
  return(c(mean,sd,n,sem))
}
