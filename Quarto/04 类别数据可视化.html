3.1 计算累积百分比

Code 
# 计算累积百分比
d$累积百分比 <- round(cumsum(d$支出金额)/sum(d$支出金额),4)*100
Code 
ggplot(d,aes(x=支出项目,y=支出金额))+
  geom_col()+
  geom_text(aes(label = 支出金额),nudge_y = 1000)

4.3 百分比条形图

Code

  geom_bar()
Code 
data3_1 %>% select(满意度,性别) %>% table() %>% as.data.frame() %>%
  group_by(满意度) %>% 
  mutate(percent=Freq/sum(Freq)) %>% ungroup() %>% 
  ggplot(aes(x=满意度,y=percent,fill=性别))+
  geom_bar(stat = "identity")
